# Sleeping in a bed against BDS: `python tools/bds-sweep/script.py
# tools/bds-sweep/scripts/sleeping.txt --out DIR` (plain BDS, flat world, an
# operator). The bed lies along +x: foot (1,-60,0), head (2,-60,0)
# (direction 3; `/setblock` of the half at (2,-60,0) places the head there
# and the foot west of it). The player stands at (0.5,-60,0.5) facing east
# and clicks the top face (1) of the head or the foot with an empty hand. Every
# scenario starts from the player back at its spot, the bed free and clear
# weather. Ticks are the client's input ticks.
21 /gamerule domobspawning false
22 /gamerule randomtickspeed 0
23 /gamerule doweathercycle false
24 /gamerule dodaylightcycle true
25 /time set 6000
26 /fill -12 -61 -12 16 -61 12 stone

29 /clear @s
30 /tp @s 0.5 -60 0.5 -90 0
31 /setblock 2 -60 0 bed ["direction"=3,"head_piece_bit"=false]
33 /testforblock 1 -60 0 bed ["direction"=3,"head_piece_bit"=false,"occupied_bit"=false]
34 /testforblock 2 -60 0 bed ["direction"=3,"head_piece_bit"=true,"occupied_bit"=false]
