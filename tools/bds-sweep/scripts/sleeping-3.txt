# Sleeping, part 3, against BDS with a second player: `python
# tools/bds-sweep/script.py tools/bds-sweep/scripts/sleeping-3.txt --observer
# tools/bds-sweep/scripts/sleeping-2-observer.txt --out DIR`. Bed A as in
# sleeping.txt: foot (1,-60,0), head (2,-60,0), direction 3. The player
# stands at (0.5,-60,0.5) facing east; the observer stays at
# (0.5,-60,6.5), awake. Ticks are the player's input ticks.
20 /gamerule domobspawning false
21 /gamerule randomtickspeed 0
22 /gamerule doweathercycle false
23 /gamerule dodaylightcycle false
24 /time set 6000
25 /fill -16 -61 -16 30 -61 16 stone

30 /tp BDSObserver 0.5 -60 6.5 -90 0
31 /setblock 2 -60 0 bed ["direction"=3,"head_piece_bit"=false]
32 /gamerule playerssleepingpercentage 50
