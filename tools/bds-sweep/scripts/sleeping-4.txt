# Sleeping, part 4, against BDS: where a sleeper stands up. `python
# tools/bds-sweep/script.py tools/bds-sweep/scripts/sleeping-4.txt --out
# DIR`. Bed A as in sleeping.txt (foot (1,-60,0), head (2,-60,0),
# direction 3); bed C lies south: foot (8,-60,0), head (8,-60,1)
# (direction 0). Each sleeper leaves its bed and stays 20 inputs before
# the next teleport, so BDS's correction is not cut short. Ticks are the
# client's input ticks.
20 /gamerule domobspawning false
21 /gamerule randomtickspeed 0
22 /gamerule doweathercycle false
23 /gamerule dodaylightcycle false
24 /time set 18000
25 /fill -12 -61 -12 16 -61 12 stone
26 /fill -12 -60 -12 16 -50 12 air
27 /gamemode survival @s
28 /clear @s
29 /tp @s 0.5 -60 0.5 -90 0
30 /setblock 2 -60 0 bed ["direction"=3,"head_piece_bit"=false]
31 /setblock 8 -60 1 bed ["direction"=0,"head_piece_bit"=false]

# 1. Bed A from the west, the head's south side walled (two blocks).
40 /fill 2 -60 1 2 -59 1 stone
42 use-on 2 -60 0 1
45 stop-sleeping
70 /tp @s 0.5 -60 0.5 -90 0

# 2. Also the foot's south side walled.
80 /fill 1 -60 1 1 -59 1 stone
82 use-on 2 -60 0 1
85 stop-sleeping
110 /tp @s 0.5 -60 0.5 -90 0

# 3. The north side too: every side of the bed walled.
120 /fill 1 -60 -1 2 -59 -1 stone
122 use-on 2 -60 0 1
125 stop-sleeping
150 /tp @s 0.5 -60 0.5 -90 0
151 /fill 1 -60 -1 2 -59 1 air
152 /setblock 2 -60 0 bed ["direction"=3,"head_piece_bit"=false]

# 4. Only one block on the head's south side (the lower one).
160 /setblock 2 -60 1 stone
162 use-on 2 -60 0 1
165 stop-sleeping
190 /tp @s 0.5 -60 0.5 -90 0
191 /setblock 2 -60 1 air

# 5. Bed A from the east of its head (in line with the bed).
200 /tp @s 3.5 -60 0.5 90 0
202 use-on 2 -60 0 1
205 stop-sleeping
230 /tp @s 0.5 -60 0.5 -90 0

# 6. Bed C (lying south) from the west of its foot, then from the east.
240 /tp @s 6.5 -60 0.5 -90 0
242 use-on 8 -60 1 1
245 stop-sleeping
270 /tp @s 9.5 -60 0.5 90 0
272 use-on 8 -60 1 1
275 stop-sleeping
300 /tp @s 0.5 -60 0.5 -90 0

# 7. A teleport wakes a sleeper.
310 use-on 2 -60 0 1
330 /tp @s 0.5 -60 0.5 -90 0
360 exit
