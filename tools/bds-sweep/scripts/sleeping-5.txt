# Sleeping, part 5, against BDS with a second player: `python
# tools/bds-sweep/script.py tools/bds-sweep/scripts/sleeping-5.txt --observer
# tools/bds-sweep/scripts/sleeping-2-observer.txt --out DIR`. Bed A as in
# sleeping.txt (head (2,-60,0)), bed B beside it (head (2,-60,4)). The
# observer stands at (0.5,-60,4.5). Ticks are the player's input ticks.
20 /gamerule domobspawning false
21 /gamerule randomtickspeed 0
22 /gamerule doweathercycle false
23 /gamerule dodaylightcycle true
24 /time set 14000
25 /fill -12 -61 -12 16 -61 12 stone
26 /fill -12 -60 -12 16 -50 12 air
27 /gamemode survival @a
28 /clear @s
29 /tp @s 0.5 -60 0.5 -90 0
30 /tp BDSObserver 0.5 -60 4.5 -90 0
31 /setblock 2 -60 0 bed ["direction"=3,"head_piece_bit"=false]
32 /setblock 2 -60 4 bed ["direction"=3,"head_piece_bit"=false]
33 /gamerule playerssleepingpercentage 50

# 1. Day comes while the player sleeps: /time set 1000 twenty inputs in.
40 /time set 14000
42 use-on 2 -60 0 1
62 /time set 1000
100 /time query daytime
101 /tp @s 0.5 -60 0.5 -90 0

# 2. Damage while asleep.
110 /time set 14000
112 use-on 2 -60 0 1
132 /damage @s 1
160 /tp @s 0.5 -60 0.5 -90 0

# 3. Half the players asked for, both asleep: the player, then the
#    observer 50 inputs later. When does the morning come?
170 /time set 14000
172 use-on 2 -60 0 1
222 /tellraw BDSObserver {"rawtext":[{"text":"bds-sweep:use-on 2 -60 4 1"}]}
400 /time query daytime
401 /tp @s 0.5 -60 0.5 -90 0
402 /tp BDSObserver 0.5 -60 4.5 -90 0
420 exit
